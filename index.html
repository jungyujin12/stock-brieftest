const CORS_HEADERS = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization', }; export default { async fetch(request, env, ctx) { if (request.method === 'OPTIONS') { return new Response(null, { headers: CORS_HEADERS }); } const url = new URL(request.url); const path = url.pathname; try { // ── 1. 네이버 뉴스 검색 if (path === '/news') { const query = url.searchParams.get('q') || ''; const naverRes = await fetch( `https://openapi.naver.com/v1/search/news.json?query=${encodeURIComponent(query)}&display=15&sort=date`, { headers: { 'X-Naver-Client-Id': env.NAVER_CLIENT_ID, 'X-Naver-Client-Secret': env.NAVER_CLIENT_SECRET } } ); const data = await naverRes.json(); return new Response(JSON.stringify(data), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } // ── 2. DART 공시 검색 (종목코드로 직접 조회) if (path === '/dart/search') { const stockCode = url.searchParams.get('stock_code') || ''; if (!stockCode) { return new Response(JSON.stringify({ list: [] }), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } const dartRes = await fetch( `https://opendart.fss.or.kr/api/list.json?crtfc_key=${env.DART_API_KEY}&stock_code=${stockCode}&page_count=5&sort=date&sort_mth=desc` ); const data = await dartRes.json(); return new Response(JSON.stringify(data), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } // ── 4. Groq 분석 if (path === '/analyze' && request.method === 'POST') { const body = await request.json(); const { stockName, stockCode, sector, newsData, dartData } = body; const groqKey = env.GROQ_API_KEY; const newsItems = (newsData?.items || []).slice(0, 10); const newsText = newsItems.map((n, i) => `[뉴스${i+1}] ${n.title?.replace(/<[^>]+>/g, '')}\n 날짜: ${n.pubDate}\n 링크: ${n.originallink || n.link}\n 내용: ${n.description?.replace(/<[^>]+>/g, '')}` ).join('\n\n'); // 링크 맵 (뉴스 번호 → 링크) const linkMap = newsItems.map(n => n.originallink || n.link); const dartText = (dartData?.list || []).slice(0, 5).map((d, i) => `[공시${i+1}] ${d.report_nm} (${d.rcept_dt}) - ${d.corp_name}` ).join('\n'); const prompt = `당신은 한국 주식 전문 애널리스트입니다. 분석 종목: ${stockName}${stockCode ? ` (${stockCode})` : ''}${sector ? ` [업종: ${sector}]` : ''} 분석일: ${new Date().toLocaleDateString('ko-KR')} === 실시간 네이버 뉴스 === ${newsText || '뉴스 없음'} === DART 최신 공시 === ${dartText || '공시 없음'} 위 실제 데이터를 바탕으로 JSON 형식으로만 응답하세요. JSON 외 텍스트 절대 금지. { "news": [ { "headline": "뉴스 제목 (원문 기반)", "desc": "2~3문장 요약", "tag": "출처/날짜", "newsIndex": 1 }, { "headline": "...", "desc": "...", "tag": "...", "newsIndex": 2 }, { "headline": "...", "desc": "...", "tag": "...", "newsIndex": 3 } ], "pros": [ { "headline": "호재 제목", "desc": "구체적 근거 포함", "tag": "호재" }, { "headline": "...", "desc": "...", "tag": "..." } ], "cons": [ { "headline": "악재 제목", "desc": "구체적 근거 포함", "tag": "악재" }, { "headline": "...", "desc": "...", "tag": "..." } ], "disclosures": [ { "headline": "공시 제목", "desc": "공시 의미와 주가 영향", "tag": "공시" } ], "summary": "팩트 중심 한줄 요약 (투자 권유 없이)" } 호재 기준: 실적개선, 수주, 신제품, 파트너십, 수출증가, 정부지원 악재 기준: 실적악화, 소송, 규제, 경쟁심화, 원가상승, 수요감소 중립 뉴스는 news에만 포함, pros/cons 금지 없는 내용 지어내기 절대 금지, 불확실하면 "확인 필요" 투자 권유 금지`; const groqRes = await fetch('https://api.groq.com/openai/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${groqKey}` }, body: JSON.stringify({ model: 'llama-3.3-70b-versatile', messages: [{ role: 'user', content: prompt }], max_tokens: 2048, temperature: 0.2, }), }); const groqData = await groqRes.json(); // 뉴스 링크 매핑 try { const raw = groqData.choices?.[0]?.message?.content || ''; const jsonMatch = raw.match(/\{[\s\S]*\}/); if (jsonMatch) { const parsed = JSON.parse(jsonMatch[0]); if (parsed.news) { parsed.news = parsed.news.map(n => ({ ...n, link: n.newsIndex ? linkMap[n.newsIndex - 1] || '' : '' })); } // 수정된 JSON으로 교체 groqData.choices[0].message.content = JSON.stringify(parsed); } } catch(e) { /* 링크 매핑 실패해도 계속 */ } return new Response(JSON.stringify(groqData), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } return new Response(JSON.stringify({ status: 'StockBrief API OK' }), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } }, };
