const CORS_HEADERS = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization', }; export default { async fetch(request, env, ctx) { if (request.method === 'OPTIONS') { return new Response(null, { headers: CORS_HEADERS }); } const url = new URL(request.url); const path = url.pathname; try { // ── 1. 네이버 뉴스 검색 ────────────────────────────── if (path === '/news') { const query = url.searchParams.get('q') || ''; const naverRes = await fetch( `https://openapi.naver.com/v1/search/news.json?query=${encodeURIComponent(query)}&display=10&sort=date`, { headers: { 'X-Naver-Client-Id': env.NAVER_CLIENT_ID, 'X-Naver-Client-Secret': env.NAVER_CLIENT_SECRET, }, } ); const data = await naverRes.json(); return new Response(JSON.stringify(data), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } // ── 2. DART 공시 검색 ──────────────────────────────── if (path === '/dart') { const corpCode = url.searchParams.get('corp_code') || ''; const dartRes = await fetch( `https://opendart.fss.or.kr/api/list.json?crtfc_key=${env.DART_API_KEY}&corp_code=${corpCode}&page_count=5&sort=date&sort_mth=desc`, ); const data = await dartRes.json(); return new Response(JSON.stringify(data), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } // ── 3. DART 기업코드 검색 ──────────────────────────── if (path === '/dart/company') { const name = url.searchParams.get('name') || ''; const dartRes = await fetch( `https://opendart.fss.or.kr/api/company.json?crtfc_key=${env.DART_API_KEY}&corp_name=${encodeURIComponent(name)}&page_count=1`, ); const data = await dartRes.json(); return new Response(JSON.stringify(data), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } // ── 4. Groq 분석 (뉴스+공시 데이터 받아서 분석) ────── if (path === '/analyze' && request.method === 'POST') { const body = await request.json(); const { stockName, stockCode, newsData, dartData } = body; const groqKey = env.GROQ_API_KEY; const newsText = (newsData?.items || []).slice(0, 8).map((n, i) => `[뉴스${i+1}] ${n.title?.replace(/<[^>]+>/g, '')} 출처: ${n.originallink || n.link} 날짜: ${n.pubDate} 내용: ${n.description?.replace(/<[^>]+>/g, '')}` ).join('\n\n'); const dartText = (dartData?.list || []).slice(0, 5).map((d, i) => `[공시${i+1}] ${d.report_nm} (${d.rcept_dt})` ).join('\n'); const prompt = `당신은 한국 주식 전문 애널리스트입니다. 분석 종목: ${stockName}${stockCode ? ` (${stockCode})` : ''} 분석일: ${new Date().toLocaleDateString('ko-KR')} === 실시간 네이버 뉴스 === ${newsText || '뉴스 없음'} === DART 최신 공시 === ${dartText || '공시 없음'} 위 실제 데이터를 바탕으로 아래 JSON 형식으로만 응답하세요. { "news": [ { "headline": "뉴스 제목", "desc": "2~3문장 요약", "tag": "출처/날짜" }, { "headline": "...", "desc": "...", "tag": "..." }, { "headline": "...", "desc": "...", "tag": "..." } ], "pros": [ { "headline": "호재 제목", "desc": "근거 포함 설명", "tag": "호재" }, { "headline": "...", "desc": "...", "tag": "..." } ], "cons": [ { "headline": "악재 제목", "desc": "근거 포함 설명", "tag": "악재" }, { "headline": "...", "desc": "...", "tag": "..." } ], "disclosures": [ { "headline": "공시 제목", "desc": "공시 의미 설명", "tag": "공시" } ], "summary": "한줄 요약 (팩트 중심, 투자 권유 없이)" } 규칙: - 제공된 뉴스/공시 데이터만 기반으로 작성 - 없는 내용 지어내기 금지, 없으면 "확인 필요" - 투자 권유 금지 - JSON만 출력`; const groqRes = await fetch('https://api.groq.com/openai/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${groqKey}`, }, body: JSON.stringify({ model: 'llama-3.3-70b-versatile', messages: [{ role: 'user', content: prompt }], max_tokens: 2048, temperature: 0.2, }), }); const groqData = await groqRes.json(); return new Response(JSON.stringify(groqData), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } return new Response(JSON.stringify({ status: 'StockBrief API OK' }), { headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' }, }); } }, };
